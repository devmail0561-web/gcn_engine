 Chargement des ressources lexicales

 LexicalResources::load(dir) appelle Lexicon::load_from_dir(dir) → std::fs::read_dir(dir).
 - Si le répertoire n'existe pas → KnowledgeError::Io → SpanishParser::new() échoue.
 - Si le répertoire existe mais est vide → HashMap vide → toutes les tables (causal_markers, causal_verb_relations, etc.) sont vides → annotateur produit des nœuds mais pas d'arêtes.
 - Conséquence : les répertoires gcn-references/taxonomies/es/ et gcn-references/taxonomies/de/ doivent exister (vides suffit pour que le crate compile et les tests passent avec assertions sur les nœuds).

 gcn-frontend-code : pattern d'ajout d'un langage

 1. Cargo.toml : ajouter tree-sitter-<lang>
 2. gcn-references/taxonomies/<lang>/<lang>_ast.yaml : fichier YAML avec sections transparent et mappings (kind → node_type, edge_type, label_strategy)
 3. resources.rs : ajouter load_<lang>() qui appelle Self::load_from(&taxonomies_root.join("<lang>/<lang>_ast.yaml"))
 4. <lang>.rs : nouveau parseur (bloc container + walk fonctions + dispatch)
 5. lib.rs : mod <lang>; + pub fn parse_<lang>(...)

     let f = form.to_lowercase();

 pub fn lemmatize_verb(form: &str) -> String {
     let f = form.to_lowercase();
     // Gérondif → infinitif -ar
     if let Some(s) = f.strip_suffix("ando") { return format!("{}ar", s); }
     // Gérondif -er/-ir
     if let Some(s) = f.strip_suffix("iendo") { return format!("{}er", s); }
     // Participe passé
     if let Some(s) = f.strip_suffix("ado") { return format!("{}ar", s); }
     if let Some(s) = f.strip_suffix("ido") { return format!("{}ir", s); }
     // Imparfait -ar
     for suf in &["aba","abas","ábamos","aban"] {
         if let Some(s) = f.strip_suffix(suf) { return format!("{}ar", s); }
     }
     // Imparfait -er/-ir
     for suf in &["ía","ías","íamos","ían"] {
         if let Some(s) = f.strip_suffix(suf) { return s.to_string(); }
     }
     // Prétérit -ar
     if let Some(s) = f.strip_suffix('ó') { return format!("{}ar", s); }
     if let Some(s) = f.strip_suffix("aron") { return format!("{}ar", s); }
     // Prétérit -ir/-er
     if let Some(s) = f.strip_suffix("ió") { return format!("{}ir", s); }
     if let Some(s) = f.strip_suffix("ieron") { return format!("{}ir", s); }
     // Présent : strip désinences
     for suf in &["amos","áis","an","as","a","o"] {
         if let Some(s) = f.strip_suffix(suf) {
             if s.len() > 2 { return format!("{}ar", s); }
         }
     }
     f.to_string()
 }

 pub fn looks_like_verb_morphologically(form: &str) -> bool {
     let f = form.to_lowercase();
     f.ends_with("ar") || f.ends_with("er") || f.ends_with("ir")
         || f.ends_with("ando") || f.ends_with("iendo")
         || f.ends_with("ado") || f.ends_with("ido")
         || f.ends_with('ó') || f.ends_with("aba") || f.ends_with("ía")
 }

 rules.rs — morphologie DE :
 pub fn is_past_tense(form: &str) -> bool {
     let f = form.to_lowercase();
     // Prétérit faible : -te -ten -test -tet
     f.ends_with("te") || f.ends_with("ten") || f.ends_with("test") || f.ends_with("tet")
     // Participe II : -t -et -en (heuristique grossière — suffisant pour bootstrap)
         || (f.ends_with('t') && f.len() > 3)
 }

 pub fn lemmatize_verb(form: &str) -> String {
     let f = form.to_lowercase();
     // Participe II : ge- préfixe + -t/-et/-en suffixe
     if f.starts_with("ge") {
         for suf in &["et","t","en"] {
             if let Some(s) = f.strip_prefix("ge").and_then(|s| s.strip_suffix(suf)) {
                 return format!("{}en", s);
             }
         }
     }
     // Prétérit faible : -te/-ten → infinitif
     if let Some(s) = f.strip_suffix("ten") { return format!("{}en", s); }
     if let Some(s) = f.strip_suffix("te")  { return format!("{}en", s); }
     // Infinitif déjà : -en/-ern → retour direct
     if f.ends_with("en") || f.ends_with("ern") { return f.to_string(); }
     // Présent : -st/-t (2e/3e pers. sing.)
     if let Some(s) = f.strip_suffix("est") { return format!("{}en", s); }
     if let Some(s) = f.strip_suffix("st")  { return format!("{}en", s); }
     if let Some(s) = f.strip_suffix('t')  {
         if s.len() > 2 { return format!("{}en", s); }
     }
     f.to_string()
 }

 pub fn looks_like_verb_morphologically(form: &str) -> bool {
     let f = form.to_lowercase();
     f.ends_with("en") || f.ends_with("ern")
         || f.ends_with("te") || f.ends_with("ten")
         || f.ends_with("end") // présent participial
         || f.starts_with("ge") // participe II
 }
